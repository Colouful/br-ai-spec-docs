## 5.3.5 BoundaryResolver 伪代码

```ts
export class BoundaryResolver {
  async resolve(rootDir: string): Promise<WorkspaceFacts> {
    if (await exists(`${rootDir}/pnpm-workspace.yaml`)) {
      return new NodeWorkspaceResolver().resolvePnpm(rootDir);
    }

    if (await hasPackageJsonWorkspaces(rootDir)) {
      return new NodeWorkspaceResolver().resolvePackageWorkspaces(rootDir);
    }

    if (await exists(`${rootDir}/lerna.json`)) {
      return new NodeWorkspaceResolver().resolveLerna(rootDir);
    }

    if (await isMavenMultiModule(rootDir)) {
      return new JavaWorkspaceResolver().resolveMaven(rootDir);
    }

    if (await exists(`${rootDir}/settings.gradle`) || await exists(`${rootDir}/settings.gradle.kts`)) {
      return new JavaWorkspaceResolver().resolveGradle(rootDir);
    }

    if (await exists(`${rootDir}/go.work`)) {
      return new GoWorkspaceResolver().resolve(rootDir);
    }

    return new FallbackManifestResolver().resolve(rootDir);
  }
}
```

### NextJsDetector 骨架

```ts
export class NextJsDetector implements TechDetector {
  name = 'NextJsDetector';

  detect(facts: ProjectFacts): DetectionResult | null {
    let score = 0;
    const reasons: string[] = [];
    const deps = facts.dependencies;

    if (deps.next) {
      score += 50;
      reasons.push('检测到 next 依赖：+50');
    }

    if (deps.react && deps['react-dom']) {
      score += 10;
      reasons.push('检测到 react 与 react-dom：+10');
    }

    if (facts.keyPaths.has('next.config.js') || facts.keyPaths.has('next.config.mjs')) {
      score += 25;
      reasons.push('检测到 next.config 配置文件：+25');
    }

    if (facts.keyPaths.has('src/app/layout.tsx')) {
      score += 15;
      reasons.push('检测到 App Router 目录：+15');
    }

    if (facts.keyPaths.has('src/pages/_app.tsx')) {
      score += 10;
      reasons.push('检测到 Pages Router 目录：+10');
    }

    if (score < 50) return null;

    return {
      detector: this.name,
      domain: 'frontend',
      language: deps.typescript ? ['TypeScript'] : ['JavaScript'],
      frameworks: ['Next.js', 'React'],
      buildTool: 'Next.js',
      confidence: Math.min(score, 100),
      tags: this.extractTags(facts),
      manifestSlug: 'frontend-react-nextjs-standard',
      reasons
    };
  }

  private extractTags(facts: ProjectFacts): string[] {
    const tags: string[] = [];
    if (facts.keyPaths.has('src/app/layout.tsx')) tags.push('app-router');
    if (facts.dependencies.tailwindcss) tags.push('tailwind');
    if (facts.dependencies.prisma) tags.push('prisma');
    return tags;
  }
}
```

### SpringBootDetector 骨架

```ts
export class SpringBootDetector implements TechDetector {
  name = 'SpringBootDetector';

  detect(facts: ProjectFacts): DetectionResult | null {
    let score = 0;
    const reasons: string[] = [];
    const text = `${facts.pomXmlText ?? ''}\n${facts.gradleText ?? ''}`;

    if (facts.manifestFiles?.pomXml || facts.keyPaths.has('pom.xml')) {
      score += 20;
      reasons.push('检测到 Maven pom.xml：+20');
    }

    if (text.includes('org.springframework.boot')) {
      score += 35;
      reasons.push('检测到 org.springframework.boot：+35');
    }

    if (text.includes('spring-boot-starter-web')) {
      score += 25;
      reasons.push('检测到 spring-boot-starter-web：+25');
    }

    if (facts.keyPaths.has('src/main/resources/application.yml') || facts.keyPaths.has('src/main/resources/application.properties')) {
      score += 10;
      reasons.push('检测到 Spring 配置文件：+10');
    }

    if (facts.keyPaths.has('src/main/java')) {
      score += 10;
      reasons.push('检测到 Java 源码目录：+10');
    }

    if (score < 50) return null;

    const tags: string[] = [];
    if (text.includes('mybatis-spring-boot-starter')) tags.push('mybatis');
    if (text.includes('spring-cloud-starter')) tags.push('spring-cloud');

    return {
      detector: this.name,
      domain: 'backend',
      language: ['Java'],
      frameworks: ['Spring Boot'],
      buildTool: facts.keyPaths.has('pom.xml') ? 'Maven' : 'Gradle',
      confidence: Math.min(score, 100),
      tags,
      manifestSlug: 'backend-java-springboot-standard',
      reasons
    };
  }
}
```

## 5.6.3 AssetCache 骨架

```ts
export class AssetCache {
  constructor(private hubClient: HubClient) {}

  async ensureAsset(input: LockedAsset): Promise<CachedAsset> {
    const cachePath = getAssetCachePath(input.checksum);

    if (await exists(cachePath)) {
      const actual = await sha256File(`${cachePath}/content.md`);
      if (actual === input.checksum) {
        return {
          hit: true,
          cachePath: `${cachePath}/content.md`,
          checksum: input.checksum
        };
      }
    }

    const content = await this.hubClient.getAssetContent({
      slug: input.slug,
      version: input.version
    });

    const actual = sha256(content.content);
    if (actual !== input.checksum) {
      throw new AiSpecError('ASSET_CHECKSUM_MISMATCH', '资产 checksum 校验失败');
    }

    await writeFile(`${cachePath}/content.md`, content.content);
    await writeJson(`${cachePath}/metadata.json`, content);

    return {
      hit: false,
      cachePath: `${cachePath}/content.md`,
      checksum: input.checksum
    };
  }
}
```

```text

↓

↓

↓

↓

↓

↓
```

## 5.7.3 骨架

```ts
export class AssetTamperChecker implements IAssetTamperChecker {
  async check(input: AssetTamperCheckInput): Promise<AssetTamperCheckResult> {
    const issues: AssetTamperIssue[] = [];
    const lock = await readJson(input.lockFilePath);
    const registry = await readJson(input.registryIndexPath);

    for (const asset of lock.assets) {
      const cachePath = getAssetCachePath(asset.checksum);
      if (!(await exists(cachePath))) {
        issues.push({
          level: 'error',
          kind: asset.kind,
          slug: asset.slug,
          message: '资产缓存不存在',
          suggestion: '请执行 npx @ex/ai-spec-auto sync .'
        });
        continue;
      }

      const actualChecksum = await sha256File(`${cachePath}/content.md`);
      if (actualChecksum !== asset.checksum) {
        issues.push({
          level: 'fatal',
          kind: asset.kind,
          slug: asset.slug,
          expectedChecksum: asset.checksum,
          actualChecksum,
          message: '检测到标准资产被篡改',
          suggestion: '请执行 sync 恢复，或创建 Project Overlay'
        });
      }
    }

    validateRegistryWithLock(registry, lock, issues);
    validateOverlays(lock.overlays, issues);

    return {
      passed: issues.every((item) => item.level !== 'fatal' && item.level !== 'error'),
      issues
    };
  }
}
```

```ts
export class WorktreeManager {
  async create(input: {
    repoRoot: string;
    runId: string;
    baseBranch: string;
    branchPrefix: string;
    worktreeRoot: string;
    requirementSummary: string;
  }): Promise<WorktreeCreateResult> {
    const branchName = createBranchName({
      prefix: input.branchPrefix,
      runId: input.runId,
      title: input.requirementSummary
    });

    const worktreePath = join(
      input.worktreeRoot,
      `${input.runId}-${slugify(input.requirementSummary)}`
    );

    await execGit(input.repoRoot, ['fetch', '--all']);
    await execGit(input.repoRoot, ['branch', branchName, input.baseBranch]);
    await execGit(input.repoRoot, ['worktree', 'add', worktreePath, branchName]);

    return {
      branchName,
      worktreePath,
      baseBranch: input.baseBranch
    };
  }
}
```

```ts
    await this.eventPublisher.publishTransition(input);

      runId: input.runId,
      state: input.to,
      updatedAt: new Date().toISOString()
    };
  }
}
```

```ts
export class PrivacyFilter implements IPrivacyFilter {
  filterRuntimeEvent(input: RuntimeEventPayload): RuntimeEventPayload {
    this.assertNoSensitiveData(input);

    return {
      ...input,
      privacy: {
        sourceCodeIncluded: false,
        rawPromptIncluded: false,
        rawResponseIncluded: false,
        absolutePathIncluded: false,
        userNameIncluded: false
      }
    };
  }

  assertNoSensitiveData(input: unknown): void {
    const text = JSON.stringify(input);

    if (looksLikeSourceCode(text)) {
      throw new AiSpecError('SOURCE_UPLOAD_FORBIDDEN', '检测到疑似源码内容，已拒绝上报');
    }

    if (looksLikeSecret(text)) {
      throw new AiSpecError('PRIVACY_POLICY_VIOLATED', '检测到疑似密钥，已拒绝上报');
    }

    if (looksLikeAbsolutePath(text)) {
      throw new AiSpecError('PRIVACY_POLICY_VIOLATED', '检测到绝对路径，已拒绝上报');
    }
  }
}
```

## 5.17.3 ExecutorRunner 骨架

```ts
export class ExecutorRunner {
  async run(input: ExecutorRunInput): Promise<ExecutorExecutionResult> {
    const provider = this.registry.get(input.executor);

    if (!provider) {
      throw new AiSpecError('EXECUTOR_NOT_AVAILABLE', `未注册执行器：${input.executor}`);
    }

    const availability = await provider.checkAvailability({
      projectRoot: input.projectRoot,
      worktreePath: input.worktreePath,
      env: process.env
    });

    if (!availability.available) {
      throw new AiSpecError('EXECUTOR_NOT_AVAILABLE', availability.fixSuggestion ?? '执行器不可用');
    }

    await provider.prepare(input.prepareInput);

    return await withTimeout(
      () => provider.execute(input.executeInput),
      input.timeoutMs,
      'EXECUTOR_TIMEOUT'
    );
  }
}
```

---

## 5.17.4 Gotchas

| 问题                  | 处理                                          |
| ------------------- | ------------------------------------------- |
| Cursor 不支持 headless | 返回 interactive-required，建议切换 Codex / Claude |
| Codex CLI 不存在       | checkAvailability 失败                        |
| Claude Code 未登录     | checkAvailability 返回修复建议                    |
| 执行器输出不规范            | result-parser 转换，失败进入 diagnosing            |
| 执行器卡死               | timeout                                     |
| 执行器重复失败             | retry 后熔断                                   |

# 5.18 P2-2：Codex Provider

## 5.18.1 适用场景

```text
结构化任务
自动化开发
远程任务调度
长任务执行
批量改造
```

---

## 5.18.2 实现要点

```text
1. 检测 codex 命令是否可用。
2. 准备 executor-input.json。
3. 准备中文 instructions.md。
4. 调用 codex 执行。
5. 解析输出成标准 ExecutionResult。
```

---

## 5.18.3 骨架

```ts
export class CodexExecutorProvider implements IExecutorProvider {
  readonly name = 'codex' as const;
  readonly displayName = 'Codex IDE';
  readonly capabilities: ExecutorCapability[] = [
    'read',
    'write',
    'shell',
    'test',
    'headless',
    'long-running',
    'structured-output'
  ];

  async checkAvailability(): Promise<ExecutorAvailabilityResult> {
    const result = await commandExists('codex');
    if (!result) {
      return {
        available: false,
        reason: '未检测到 codex 命令',
        fixSuggestion: '请安装 Codex IDE CLI，或切换为 cursor / claude-code。'
      };
    }
    return { available: true };
  }

  async prepare(input: ExecutorPrepareInput): Promise<ExecutorPrepareResult> {
    const dir = `.codex/tmp/${input.runId}`;
    await writeJson(`${dir}/executor-input.json`, sanitizeExecutorInput(input));
    await writeMarkdown(`${dir}/instructions.md`, renderCodexInstructions(input));
    return {
      prepared: true,
      executorInputPath: `${dir}/executor-input.json`,
      instructionFilePath: `${dir}/instructions.md`
    };
  }

  async execute(input: ExecutorExecuteInput): Promise<ExecutorExecutionResult> {
    const result = await runProcess({
      command: 'codex',
      args: ['run', '--input', `.codex/tmp/${input.runId}/executor-input.json`],
      cwd: input.worktreePath,
      timeoutMs: input.timeoutMs
    });

    return parseCodexResult(result);
  }
}
```

# 5.19 P2-3：Cursor Provider

## 5.19.1 适用场景

```text
本地人工辅助开发
IDE 内协作
开发者边看边改
交互式需求
```

---

## 5.19.2 约束

```text
Cursor 不应作为 remote-orchestrated 默认执行器。
如果当前环境不支持 headless 调用，应返回需要人工交互。
```

---

## 5.19.3 骨架

```ts
export class CursorExecutorProvider implements IExecutorProvider {
  readonly name = 'cursor' as const;
  readonly displayName = 'Cursor';
  readonly capabilities: ExecutorCapability[] = [
    'read',
    'write',
    'interactive',
    'structured-output'
  ];

  async checkAvailability(): Promise<ExecutorAvailabilityResult> {
    const exists = await commandExists('cursor');
    return exists
      ? { available: true }
      : {
          available: false,
          reason: '未检测到 cursor 命令',
          fixSuggestion: '请安装 Cursor 命令行工具，或切换执行器。'
        };
  }

  async prepare(input: ExecutorPrepareInput): Promise<ExecutorPrepareResult> {
    await ensureCursorRule(input.projectRoot);
    const taskFile = `.cursor/tmp/${input.runId}/task.md`;
    await writeMarkdown(taskFile, renderCursorTask(input));
    return {
      prepared: true,
      instructionFilePath: taskFile
    };
  }

  async execute(input: ExecutorExecuteInput): Promise<ExecutorExecutionResult> {
    if (!supportsCursorHeadless()) {
      return {
        success: false,
        status: 'failed',
        summary: '当前 Cursor 环境不支持无头执行，需要人工在 IDE 中继续。',
        changedFiles: [],
        riskList: [
          {
            level: 'warning',
            message: 'Cursor 更适合人工辅助模式，远程自动化建议使用 Codex 或 Claude Code。'
          }
        ],
        error: {
          code: 'EXECUTOR_PERMISSION_DENIED',
          message: '当前环境不支持 Cursor 无头执行'
        }
      };
    }

    return runCursorCommand(input);
  }
}
```


# 5.25 本章节实现边界

完成第五部分后，AI 编程助手必须严格按以下顺序实施：

```text
1. 先做配置系统。
2. 再做扫描引擎。
3. 再做 init plan 和写入轻量文件。
4. 再做全局缓存和 Hub sync。
5. 再做资产防篡改。
6. 再做 worktree。
7. 再做状态机。
8. 再做 Context Builder。
9. 再做 Visual 上报。
10. 再做 Hub 资产 API。
11. 再做 Agent Profile。
12. 再做 Codex / Cursor / Claude Code Provider。
13. 最后再做 Asset Factory / Coze / 远程入口。
```

禁止行为：

```text
1. 禁止扫描阶段写业务项目。
2. 禁止绕过 lock 使用资产。
3. 禁止直接修改 Hub 已发布资产。
4. 禁止把 Codex / Cursor / Claude Code 写死为唯一执行器。
5. 禁止上传目标项目源码。
6. 禁止状态机无限重试。
7. 禁止不创建 branch / worktree 直接改当前工作区。
```
